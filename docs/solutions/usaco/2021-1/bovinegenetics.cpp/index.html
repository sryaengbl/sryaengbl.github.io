#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#ifdef LOCAL
#include "algo/debug.h"
#else
#define dbg(...) 0
#define dbgn(...) 0
#endif

const int N = 1e5 + 5;
const ll MOD = 1e9 + 7;

int n;
ll dp[N][4][4][4];
ll pref[4][4];
char mp[4] = {'A', 'T', 'C', 'G'};

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    string str;
    cin >> str;
    n = str.size();
    for (int a = 0; a < 4; a++) {
        for (int b = 0; b < 4; b++) {
            if (str[0] != mp[b] && str[0] != '?') continue;
            dp[1][a][b][b] = 1;
        }
    }
    for (int i = 1; i < n; i++) {
        memset(pref, 0, sizeof(pref));
        for (int a = 0; a < 4; a++) {
            for (int b = 0; b < 4; b++) {
                for (int c = 0; c < 4; c++) {
                    for (int d = 0; d < 4; d++) {
                        if (str[i] != mp[d] && str[i] != '?') continue;
                        if (c != d) {
                            dp[i + 1][a][b][d] = (dp[i + 1][a][b][d] + dp[i][a][b][c]) % MOD;
                        }
                        if (a == c) {
                            pref[b][d] = (pref[b][d] + dp[i][a][b][c]) % MOD;
                        }
                    }
                }
            }
        }
        for (int b = 0; b < 4; b++) {
            for (int d = 0; d < 4; d++) {
                dp[i + 1][b][d][d] = (dp[i + 1][b][d][d] + pref[b][d]) % MOD;
            }
        }
    }
    ll ans = 0;
    for (int a = 0; a < 4; a++) {
        for (int b = 0; b < 4; b++) {
            ans = (ans + dp[n][a][b][a]) % MOD;
        }
    }
    cout << ans << '\n';
    return 0;
}
