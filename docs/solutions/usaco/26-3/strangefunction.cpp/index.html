#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#ifdef LOCAL
#include "algo/debug.h"
#else
#define dbg(...) 0
#define dbgn(...) 0
#endif

ll MOD = 1e9 + 7;

void solve() {
    string str;
    cin >> str;
    reverse(str.begin(), str.end());
    ll c = 1, val = 0;
    bool form = 1, ez = 1;
    for (int i = 0; i < int(str.size()); i++) {
        if (str[i] - '0' > 1) form = 0;
        if ((str[i] - '0') & 1) {
            if (i == 0) ez = 0;
            val += c;
            val %= MOD;
        }
        c = (c * 2) % MOD;
    }
    if (ez) {
        val = val * 500000004 % MOD * 3 % MOD;
    } else {
        val -= 1;
        val = val * 500000004 % MOD * 3 % MOD;
        val++;
    }
    if (!form) val++;
    cout << val << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
