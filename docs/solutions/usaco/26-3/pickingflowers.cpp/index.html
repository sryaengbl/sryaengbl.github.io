#include <bits/stdc++.h>

void solve() {
    int n, m, k, l;
    cin >> n >> m >> k >> l;
    vector<vector<int>> adj(n);
    vector<int> f(k), d(l);
    for (int i = 0; i < k; i++) {
        cin >> f[i];
        --f[i];
    }
    for (int i = 0; i < l; i++) {
        cin >> d[i];
        --d[i];
    }
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        --a, --b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    vector<int> dist(n, 1e9);
    queue<int> q;
    dist[0] = 0;
    q.push(0);
    while (!q.empty()) {
        int c = q.front();
        q.pop();
        for (int x : adj[c]) {
            if (dist[x] == 1e9) {
                dist[x] = dist[c] + 1;
                q.push(x);
            }
        }
    }
    f.push_back(0);
    sort(f.begin(), f.end(), [&](int a, int b) { return dist[a] < dist[b]; });
    for (int i = 1; i < (int)f.size(); i++) {
        if (dist[f[i]] == dist[f[i - 1]]) {
            cout << string(n - 1, '0') << '\n';
            return;
        }
    }
    vector<int> req(n, -1);
    for (int x : f) {
        req[dist[x]] = x;
    }
    vector<int> possible;
    for (int x : d) {
        if (dist[x] >= dist[f.back()]) {
            possible.push_back(x);
        }
    }
    if (possible.empty()) {
        cout << string(n - 1, '0') << '\n';
        return;
    }
    vector<bool> g(n, 0);
    g[0] = 1;
    q.push(0);
    while (!q.empty()) {
        int c = q.front();
        q.pop();
        for (int x : adj[c]) {
            if (dist[x] != dist[c] + 1) continue;
            if (req[dist[x]] != -1 && req[dist[x]] != x) continue;
            if (g[x]) continue;
            g[x] = 1;
            q.push(x);
        }
    }
    vector<bool> g2(n, 0);
    for (int x : possible) {
        if (req[dist[x]] != -1 && req[dist[x]] != x) continue;
        if (!g2[x]) {
            g2[x] = 1;
            q.push(x);
        }
    }
    while (!q.empty()) {
        int c = q.front();
        q.pop();
        for (int x : adj[c]) {
            if (dist[x] != dist[c] - 1) continue;
            if (req[dist[x]] != -1 && req[dist[x]] != x) continue;
            if (g2[x]) continue;
            g2[x] = 1;
            q.push(x);
        }
    }
    for (int i = 1; i < n; i++) {
        cout << (g[i] && g2[i] ? '1' : '0');
    }
    cout << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
