Analysis

ingenious method from editorial given a tree rooted at node 0, how many topological sorts does the tree have? it starts that there are (n-1)! permutations that serve as candidates look at the subtree of node x the probability that node x is in the front of all its subtree elements is 1/subt[x], where subt[x] is the size of the subtree the probability that this occurs for all nodes is product of 1/subt[x] for all nodes x in the tree this is because the order of the nodes is completely independent from each other in the permutation if you knew the order of all the nodes in a root's subtree (e.g. their permutation), you still won't know where to insert the root into that permutation. all possible "insertions" into the ordering are equally likely; we don't gain new information of course, this product currently includes 1/subt[0] = 1/n so we need to multiply by n because we are guaranteed node 0 is the root the number of toposorts is n! * prod of 1/subt[x] for all nodes x brilliant!

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const ll MOD = 1e9 + 7;

ll exp(ll base, ll exp) {
    ll ret = 1;
    while (exp) {
        if (exp & 1) ret = ret * base % MOD;
        base = base * base % MOD;
        exp >>= 1;
    }
    return ret;
}

void solve() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n);
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        a--, b--;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    vector<ll> fact(n + 1), inv(n + 1);
    fact[0] = 1;
    for (int i = 1; i <= n; i++) {
        fact[i] = fact[i - 1] * i % MOD;
    }
    for (int i = 1; i <= n; i++) {
        inv[i] = exp(i, MOD - 2);
    }
    vector<ll> subt(n, 1);
    ll ans = 1;
    auto dfs = [&](int u, int v, auto&& self) -> void {
        for (auto x : adj[u]) {
            if (x == v) continue;
            self(x, u, self);
            subt[u] += subt[x];
        }
        ans *= inv[subt[u]];
        ans %= MOD;
    };
    dfs(0, -1, dfs);
    ll curr = ans;
    auto dfs2 = [&](int u, int v, auto&& self) -> void {
        for (auto x : adj[u]) {
            if (x == v) continue;
            ll subtx = subt[x], subtu = subt[u], pc = curr;
            curr = curr * subt[u] % MOD;
            subt[u] -= subt[x];
            curr = curr * inv[subt[u]] % MOD;

            curr = curr * subt[x] % MOD;
            subt[x] += subt[u];
            curr = curr * inv[subt[x]] % MOD;

            ans += curr;
            self(x, u, self);
            subt[x] = subtx, subt[u] = subtu, curr = pc;
        }
    };
    dfs2(0, -1, dfs2);
    ans %= MOD;
    cout << ans * exp(fact[n - 1], MOD - 2) % MOD << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
