#include <bits/stdc++.h>

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int n, h;
    cin >> n >> h;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int k;
    cin >> k;
    vector<bool> wc(n, 0);
    for (int i = 0; i < k; i++) {
        int x;
        cin >> x;
        wc[x - 1] = 1;
    }
    priority_queue<pair<int, ll>> pq;
    for (int i = 0; i < h; i++) {
        pq.push({wc[i], -a[i]});
    }
    vector<pair<int, ll>> pref(n - h + 2);
    pref[0] = {0, 0};
    for (int i = 0; i < n - h + 1; i++) {
        auto [w, c] = pq.top();
        pq.pop();
        pref[i + 1] = pref[i];
        pref[i + 1].first += w;
        pref[i + 1].second += -c;
        pq.push({wc[h + i], -a[h + i]});
    }
    int q;
    cin >> q;
    while (q--) {
        ll s;
        cin >> s;
        ll ans = s / (pref[n - h + 1].second) * pref[n - h + 1].first;
        s %= pref[n - h + 1].second;
        int l = 0, r = n - h;
        while (l < r) {
            int m = (l + r + 1) / 2;
            if (pref[m].second <= s)
                l = m;
            else
                r = m - 1;
        }
        cout << ans + pref[l].first << '\n';
    }
    return 0;
}
