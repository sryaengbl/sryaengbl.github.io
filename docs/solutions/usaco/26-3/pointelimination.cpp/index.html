void solve() {
    int n;
    cin >> n;
    multiset<int> x, y;
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        x.insert(a);
        y.insert(b);
    }
    bool good = 1;
    int a = 0, b = 0, f = 0;
    while (!x.empty() && good) {
        int c = *x.begin();
        while (x.count(c) && good) {
            if (x.count(c) & 1) {
                if (!x.count(c + 1)) {
                    good = 0;
                    break;
                }
                x.extract(c + 1);
                x.extract(c);
                b++;
            } else {
                if (x.count(c + 1) >= 2) {
                    x.extract(c + 1);
                    x.extract(c + 1);
                    x.extract(c);
                    x.extract(c);
                    f += 2;
                } else {
                    x.extract(c);
                    x.extract(c);
                    a++;
                }
            }
        }
    }
    int a2 = 0, b2 = 0, f2 = 0;
    while (!y.empty() && good) {
        int c = *y.begin();
        while (y.count(c) && good) {
            if (y.count(c) & 1) {
                if (!y.count(c + 1)) {
                    good = 0;
                    break;
                }
                y.extract(c + 1);
                y.extract(c);
                b2++;
            } else {
                if (y.count(c + 1) >= 2) {
                    y.extract(c + 1);
                    y.extract(c + 1);
                    y.extract(c);
                    y.extract(c);
                    f2 += 2;
                } else {
                    y.extract(c);
                    y.extract(c);
                    a2++;
                }
            }
        }
    }
    if (!good) {
        cout << "NO\n";
        return;
    }
    for (int i = 0; i <= f; i += 2) {
        int j = (b + f) - i - a2;
        if (j >= 0 && !(j & 1) && (j <= f2)) {
            cout << "YES\n";
            return;
        }
    }
    cout << "NO\n";
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
