#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#ifdef LOCAL
#include "algo/debug.h"
#else
#endif

void solve() {
    int n, d = 2;
    cin >> n;
    set<int> uq;
    vector<vector<int>> a(n, vector<int>(d));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < d; j++) {
            cin >> a[i][j];
            uq.insert(a[i][j]);
            uq.insert(a[i][j] + 1);
        }
    }
    vector<int> cc(uq.begin(), uq.end());
    int mx = uq.size();
    vector<vector<int>> v(d, vector<int>(mx));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < d; j++) {
            a[i][j] = lower_bound(cc.begin(), cc.end(), a[i][j]) - cc.begin();
            v[j][a[i][j]]++;
        }
    }
    bool good = 1;
    vector<int> f(d, 0), tt(d);
    for (int i = 0; i < d && good; i++) {
        vector<int> s(mx), t(mx);

        for (int j = 0; j < mx && good; j++) {
            int c = v[i][j];
            for (int k = 1; k < d; k++) {
                if (j - k >= 0) c -= t[j - k];
            }
            if (c < 0) {
                good = 0;
                break;
            }
            t[j] = c % d;
            s[j] = (c - t[j]) / d;
        }
        if (!good) break;
        tt[i] = accumulate(t.begin(), t.end(), 0);
        for (int j = 0; j + d <= mx; j++) {
            int amt = 1e9;
            for (int k = 0; k < d; k++) {
                amt = min(amt, s[j + k]);
            }
            f[i] += amt * d;
            for (int k = 0; k < d; k++) {
                s[j + k] -= amt;
            }
        }
    }
    int lo = accumulate(tt.begin(), tt.end(), 0);
    int hi = lo + accumulate(f.begin(), f.end(), 0);
    int target = n / d;
    if (good && n % d == 0 && lo <= target && target <= hi && (target - lo) % d == 0) {
        cout << "YES\n";
    } else {
        cout << "NO\n";
    }
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
