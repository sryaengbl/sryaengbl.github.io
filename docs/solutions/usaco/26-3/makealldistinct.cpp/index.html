#include <bits/stdc++.h>
using namespace std;
using ll = long long;

#include "algo/debug.h"

#endif

void solve() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    if (k < 0) {
        for (int i = 0; i < n; i++) {
            a[i] = -a[i];
        }
        reverse(a.begin(), a.end());
        for (int i = n - 1; i >= 0; i--) {
            a[i] -= a[0] - 1;
        }
        k = -k;
    }
    vector<int> cnt(n + 1);
    for (int i = 0; i < n; i++) {
        cnt[a[i]]++;
    }
    ;
    ll ans = 0;
    for (int i = 1; i <= n - k; i++) {
        if (cnt[i] > 1) {
            ans += cnt[i] - 1;
            cnt[i + k] += cnt[i] - 1;
        }
    }
    ;
    for (int i = n - k + 1; i <= n; i++) {
        ans += (ll)(cnt[i] - 1) * (cnt[i]) / 2;
    }
    cout << ans << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
