void solve() {
    int n, m;
    cin >> n >> m;
    string t;
    cin >> t;
    vector<string> str(n);
    vector<set<pair<int, int>>> v(26);
    for (int i = 0; i < n; i++) {
        string x;
        cin >> x;
        for (int j = 0; j < m; j++) {
            v[x[j] - 'a'].insert({i, j});
        }
        str[i] = x;
    }
    vector<tuple<int, int, int, int>> ans;
    for (int i = 0; i < m; i++) {
        if (str[0][i] != t[i]) {
            int at = -1;
            for (int j = i; j < m; j++) {
                if (str[0][j] == t[i]) {
                    at = j;
                    break;
                }
            }
            if (at >= 0) {
                ans.push_back({1, 1, i + 1, at + 1});
                swap(str[0][i], str[0][at]);
            }
        }
    }
    vector<vector<int>> has(n, vector<int>(26));
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < m; j++) {
            has[i][str[i][j] - 'a']++;
        }
    }
    for (int i = 0; i < m; i++) {
        if (str[0][i] == t[i]) continue;
        bool fixed = 0;
        for (int j = 1; j < n && !fixed; j++) {
            if (has[j][t[i] - 'a']) {
                for (int k = 0; k < m && !fixed; k++) {
                    if (str[j][k] != t[i]) continue;
                    if (k != i) ans.push_back({1, j + 1, k + 1, i + 1});
                    swap(str[j][i], str[j][k]);
                    ans.push_back({2, 1, j + 1, i + 1});
                    has[j][str[j][i] - 'a']--;
                    swap(str[0][i], str[j][i]);
                    has[j][str[j][i] - 'a']++;
                    fixed = 1;
                }
            }
        }
    }
    cout << ans.size() << '\n';
    for (auto x : ans) {
        int a, b, c, d;
        tie(a, b, c, d) = x;
        cout << a << ' ' << b << ' ' << c << ' ' << d << '\n';
    }
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
