void solve() {
    int n, m;
    cin >> n >> m;
    string t;
    cin >> t;
    vector<string> str(n);
    vector<set<pair<int, int>>> v(26);
    for (int i = 0; i < n; i++) {
        string x;
        cin >> x;
        for (int j = 0; j < m; j++) {
            v[x[j] - 'a'].insert({i, j});
        }
        str[i] = x;
    }
    vector<tuple<int, int, int, int>> ans;
    for (int i = 0; i < m; i++) {
        char c = t[i];
        auto [a, b] = *v[c - 'a'].begin();
        if (a == 0) {
            if (a == 0 && b == i) {
                v[str[0][i] - 'a'].erase({0, i});
                continue;
            };
            ans.push_back({1, 1, i + 1, b + 1});
            v[str[0][i] - 'a'].erase({0, i});
            v[str[0][b] - 'a'].erase({0, b});
            swap(str[0][i], str[a][b]);
            v[str[0][b] - 'a'].insert({0, b});
        } else {
            ans.push_back({1, a + 1, i + 1, b + 1});
            v[str[a][b] - 'a'].erase({a, b});
            v[str[a][i] - 'a'].erase({a, i});
            swap(str[a][b], str[a][i]);
            v[str[a][b] - 'a'].insert({a, b});
            v[str[a][i] - 'a'].insert({a, i});
            ans.push_back({2, 1, a + 1, i + 1});
            v[str[a][i] - 'a'].erase({a, i});
            v[str[0][i] - 'a'].erase({0, i});
            swap(str[a][i], str[0][i]);
            v[str[a][i] - 'a'].insert({a, i});
        }
    }
    cout << ans.size();
    if (ans.size()) cout << '\n';
    for (int i = 0; i < (int)ans.size(); i++) {
        int a, b, c, d;
        tie(a, b, c, d) = ans[i];
        cout << a << ' ' << b << ' ' << c << ' ' << d;
        if (i < (int)ans.size() - 1) cout << '\n';
    }
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
        if (tt) cout << '\n';
    }
    return 0;
}
