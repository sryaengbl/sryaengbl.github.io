it's well-known that the number of moves it takes to sort a permutation via adjacent swaps is the number of inversions we can prove that the number of inversions >= sum (p[i]-i)/2 consider the original sorted permutation. we can swap adjacent elements to make it equal to the original array a. count the number of swaps required this number of swaps is also going to be equal to the number of inversions for each individual swap, we also need to consider how (p[i]-i)/2 is affected firstly, whenever (p[i]-i)/2 increases from a swap, realize that this means the larger element of the swap was moving "down" and the smaller element was moving "up" this means that we must have gained an inversion too. so we try to build the minimum case where (p[i]-i)/2 does not equal # inversions this case is 3 2 1 why? because we have 1 2 3, f(p)=0, #i=0 2 1 3, f(p)=1, #i=1 2 3 1, f(p)=2, #i=2 3 2 1, f(p)=2, #i=3 we conjecture #i != f(p) iff 3 2 1 appears in some form proof: firstly, consider the reverse direction. it's clear that if 3 .... 2 ... 1 appears, if we take #i which is the optimal number of moves, we can swap into ...3 2 1... In this case, the only way to resolve the order is to handle (3, 2, 1) which is the faliure case of f(p). for the forward direction, assume #i=f(p) start with #1 and move it to the front. then start with #2 and move it to the front. continue this way. this is an optimal strategy because #i decreases by one each time. #i can decrease by at most 1 each swap. assume WLOG that swapping #1 with p[i] decreases #inv but not f(p) we also have p[i+1]=1 by this WLOG assumption we must have p[i] > i that means that there are n-p[i]-1 elements > p[i] however, there are only n-p[i]-2 indexes available > p[i] because p[i+1] is occupied by 1 by the pigeonhole principle, there must be an element > p[i] that occurs an element >p[i] in the indexes [0... i-1]. this means that there's a 3 2 1 in the array thus, #i=!f(p) iff 3 2 1 appears. implementation: using stacks, we can calculate l[i] and r[i]. the naive method is for (int i = 0; i < n; i++) { int L = i, R = i + n - 1; bool bad = 0; for (int j = i; j < i + n; j++) { if (l[j] >= L && r[j] <= R) bad = 1; } if (!bad) ans.push_back((n - i) % n); } however we can use math on inequalities to derive i <= min(n-1, j, l[j]) i >= max(0, n-j+1, r[j]-n+1) then we can use prefix sums to calculate which i's work. in the future, correlating (p[i]-i)/2 with #321's could be interesting what does #inv - (p[i]-i)/2 equal? instructive problem on using inequalities to reduce time complexity

void solve() {
    int n;
    cin >> n;
    vector<int> p(n), ext(2 * n);
    for (int i = 0; i < n; i++) {
        cin >> p[i];
        p[i]--;
        ext[i] = ext[i + n] = p[i];
    }
    stack<pair<int, int>> stk;
    stk.push({-1, 2 * n});
    vector<int> l(n * 2), r(n * 2);
    for (int i = 2 * n - 1; i >= 0; i--) {
        while (stk.top().first > ext[i]) stk.pop();
        r[i] = stk.top().second;
        stk.push({ext[i], i});
    }
    while (!stk.empty()) stk.pop();
    stk.push({1e9, -1});
    for (int i = 0; i < 2 * n; i++) {
        while (stk.top().first < ext[i]) stk.pop();
        l[i] = stk.top().second;
        stk.push({ext[i], i});
    }
    vector<int> pref(n, 0);
    for (int i = 0; i < 2 * n; i++) {
        int mn = max(max(0, i - n + 1), r[i] - n + 1);
        int mx = min(min(n - 1, i), l[i]);
        if (mn > mx) continue;
        if (mn < n) pref[mn]++;
        if (mx + 1 < n) pref[mx + 1]--;
    }
    vector<int> ans;
    for (int i = 0; i < n; i++) {
        if (i) pref[i] += pref[i - 1];
        if (pref[i] == 0) ans.push_back((n - i) % n);
    }
    sort(ans.begin(), ans.end());
    cout << ans.size() << '\n';
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i];
        if (i < ans.size() - 1) cout << ' ';
    }
    cout << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
