#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#ifdef LOCAL
#include "algo/debug.h"
#else
#define dbg(...) 0
#define dbgn(...) 0
#endif

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int n;
    cin >> n;
    vector<ll> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    set<int> ms = {1, n};
    for (int i = 2; i < n; i++) {
        if ((a[i] + a[i - 1] - 1) / a[i - 1] > 1) {
            ms.insert(i);
        }
    }
    auto solve = [&](__int128_t t) -> ll {
        vector<int> v(ms.begin(), ms.end());
        __int128_t fi = 1, fq = 1;
        for (int i = 0; i < (int)v.size(); i++) {
            ll ratio;
            if (v[i] == 1)
                ratio = a[1];
            else
                ratio = (a[v[i]] + a[v[i] - 1] - 1) / (a[v[i] - 1]);
            fi = fi + (ratio - 1) * fq + (i ? v[i] - v[i - 1] : 1);
            fq = (i ? ratio : a[1] + 1) * fq;
            if (fi > t) return 0;
        }
        t -= fi;
        return ((t + fq) / fq) * a[n];
    };
    int q;
    cin >> q;
    while (q--) {
        int i, v;
        ll t;
        cin >> i >> v >> t;
        ms.extract(i);
        ms.extract(i + 1);
        a[i] = v;
        if ((i == 1) || (i == n) || ((a[i] + a[i - 1] - 1) / a[i - 1] > 1)) {
            ms.insert(i);
        }
        if (i < n && ((i + 1 == n) || ((a[i + 1] + a[i] - 1) / a[i] > 1))) {
            ms.insert(i + 1);
        }
        if (ms.size() < 70)
            cout << solve(t) << '\n';
        else
            cout << 0 << '\n';
    }
    return 0;
}
