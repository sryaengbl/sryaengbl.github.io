Analysis

my first approach was to assume that every node had an equal likelihood of being =1 however, this is misguided. because the algorithm doesn't truly make a random tree, not all nodes are equally likely to be picked. then my intuition become to calculate for each node (prob that node was root) * (prob to generate this tree given this node is root) however, calculating (prof that node was root) is somewhat not-straightforward. so I scrapped these couple of ideas for now to solve a subproblem. given that node 0 is the actual root (big assumption, but useful to understand the problem) what is the probability that the tree was generated? well, given a valid ordering of the nodes to be added, the probability is 1/(n-1)! so then the solution 1/(n-1)! * (# of valid orderings) that's a very similar problem to counting the number of topological sorts so then I derived the formula for the number of topo sorts of a given tree this is done recursively. dp[x] stores the answer for the subtree rooted at [x] dp[x]=(subt[x]-1)! / (subt[c1]! * subt[c2]! * ... * subt[cm]!) * dp[c1] * ... * dp[cm] where m is the number of children and ci is the ith child but wait. once we count one toposort for a given root node, this is guaranteed to be unique, since the toposort is rooted at that specific node AND all valid solutions are counted if we run this dp for all nodes there are n! assignments of indexes. each assignment of indexes is bijective with a toposort therefore the answer is 1/(n! * (n-1)!) * # toposorts rooted at all nodes the number of toposorts at node 0 can be calculated easily. then the rest of the problem is rerooting the tree. rerooting is where most of the implementation difficulties come from but it is doable just by looking at the formulas properly. i'd like to find a way to implement the rerooting more cleanly...

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

#include "algo/debug.h"

#define dbg(...) 0
#define dbgn(...) 0
#endif

const ll MOD = 1e9 + 7;

ll exp(ll base, ll exp) {
    ll ret = 1;
    while (exp) {
        if (exp & 1) ret = ret * base % MOD;
        base = base * base % MOD;
        exp >>= 1;
    }
    return ret;
}

void solve() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n);
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        a--, b--;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    vector<ll> fact(n + 1), inv(n + 1);
    fact[0] = 1;
    for (int i = 1; i <= n; i++) {
        fact[i] = fact[i - 1] * i % MOD;
    }
    inv[n] = exp(fact[n], MOD - 2);
    for (int i = n - 1; i >= 0; i--) {
        inv[i] = inv[i + 1] * (i + 1) % MOD;
    }
    vector<ll> subt(n, 1), dp(n, 1);
    auto dfs = [&](int u, int v, auto&& self) -> void {
        for (auto x : adj[u]) {
            if (x == v) continue;
            self(x, u, self);
            subt[u] += subt[x];
            dp[u] = dp[u] * dp[x] % MOD;
            dp[u] = dp[u] * inv[subt[x]] % MOD;
        }
        dp[u] = dp[u] * fact[subt[u] - 1] % MOD;
    };
    dfs(0, -1, dfs);
    ll ans = dp[0];
    auto dfs2 = [&](int u, int v, auto&& self) -> void {
        for (auto x : adj[u]) {
            if (x == v) continue;
            ll dpu = dp[u], subtu = subt[u], dpx = dp[x], subtx = subt[x];
            dp[u] *= inv[subt[u] - 1] * fact[subt[u] - 1 - subt[x]] % MOD;
            dp[u] %= MOD;
            dp[u] *= fact[subt[x]];
            dp[u] %= MOD;
            dp[u] *= exp(dp[x], MOD - 2);
            dp[u] %= MOD;
            subt[u] -= subt[x];
            subt[x] += subt[u];
            dp[x] *= dp[u];
            dp[x] %= MOD;
            dp[x] *= fact[subt[x] - 1] * inv[subtx - 1] % MOD;
            dp[x] %= MOD;
            dp[x] *= inv[subt[u]];
            dp[x] %= MOD;
            ans += dp[x];
            self(x, u, self);
            dp[u] = dpu;
            subt[u] = subtu;
            dp[x] = dpx;
            subt[x] = subtx;
        }
    };
    dfs2(0, -1, dfs2);
    ans %= MOD;
    ans = ans * inv[n] % MOD * inv[n - 1] % MOD;
    cout << ans << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
