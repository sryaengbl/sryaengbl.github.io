#include <bits/stdc++.h>

void solve1() {
    ll cow, t;
    cin >> cow >> t;
    if (t < 2 * cow) {
        cout << cow << '\n';
        return;
    }
    if (t < 3 * cow) {
        cout << t - 2 * cow + 1 << '\n';
        return;
    }
    if (cow == 0) {
        if (t <= 3) {
            cout << (t == 2 ? 1 : 0) << '\n';
            return;
        }
        vector<ll> dp;
        dp.push_back(4);
        dp.push_back(7);
        for (int i = 0; i < 1e9; i++) {
            if ((int)dp.size() < i + 2) {
                dp.push_back(dp[dp.size() - 1] + dp[dp.size() - 2]);
            }
            if (dp[i + 1] > t) {
                cout << dp[i] / 2 - (t - dp[i]) << '\n';
                return;
            }
        }
    }
    vector<ll> dp;
    dp.push_back(3 * cow);
    dp.push_back(dp[0] + dp[0] / 2 + 1);
    for (int i = 0; i < 1e9; i++) {
        if ((int)dp.size() < i + 2) {
            dp.push_back(dp[dp.size() - 1] + dp[dp.size() - 2]);
        }
        if (dp[i + 1] > t) {
            cout << dp[i] / 2 - (t - dp[i]) << '\n';
            return;
        }
    }
}

void solve2() {}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        int t;
        cin >> t;
        if (t == 1)
            solve1();
        else
            solve2();
    }

}
