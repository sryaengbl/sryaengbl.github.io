Analysis

the graph can be decomposed into multiple trees where all nodes flow into the root of each tree, which is the party. each party is a tree to change a party from one type to another, the root of the tree changes to add a new party, a branch of an existing tree is severed and becomes independent. the operation of splitting trees is difficult to handle. however, if we reverse the queries, it becomes merging trees, which is a class dsu problem the main difficulty then becomes how to store the root of each tree properly given that asking for the representative node of a dsu doesn't necessarily return the root of the tree, but rather an arbitrary node in the tree this code solves that problem by storing the types inside the dsu too. since i'm not too familiar with modifying DSUs, this part was what consumed most of my implementation time. key takeaways: reversing the queries, storing type inside the DSU.

#include <bits/stdc++.h>
using namespace std;

struct UF {
    vector<int> e, type;

    UF(int n) : e(n, -1), type(n, -1) {}

    int size(int x) { return -e[find(x)]; }
    int find(int x) { return e[x] < 0 ? x : e[x] = find(e[x]); }

    bool join(int a, int b) {
        a = find(a), b = find(b);
        if (a == b) return false;
        if (e[a] > e[b]) swap(a, b);
        e[a] += e[b];
        e[b] = a;

        if (type[a] == -1) type[a] = type[b];
        return true;
    }

    int getType(int x) { return type[find(x)]; }

    void setType(int x, int t) { type[find(x)] = t; }
};

void solve() {
    int n;
    cin >> n;
    vector<int> f(n);
    for (int i = 0; i < n; i++) {
        cin >> f[i];
        f[i]--;
    }
    int m;
    cin >> m;
    map<char, int> mp;
    mp['C'] = 0, mp['O'] = 1, mp['W'] = 2;
    vector<pair<int, int>> upd(m);
    vector<int> t(n, -1);
    vector<vector<int>> fa(n);
    for (int i = 0; i < m; i++) {
        char c;
        cin >> upd[i].first >> c;
        upd[i].first--;
        upd[i].second = mp[c];
        t[upd[i].first] = upd[i].second;
        fa[upd[i].first].push_back(i);
    }
    UF dsu(n);
    for (int i = 0; i < n; i++) {
        if (t[i] == -1) {
            dsu.join(i, f[i]);
        }
    }
    vector<vector<int>> ans(m + 1, vector<int>(3));
    for (int i = 0; i < n; i++) {
        if (t[i] != -1) {
            ans[m][t[i]] += dsu.size(i);
            dsu.setType(i, t[i]);
        }
    }
    for (int i = m - 1; i >= 0; i--) {
        ans[i] = ans[i + 1];
        int idx = upd[i].first;
        if (fa[idx][0] == i) {
            ans[i][upd[i].second] -= dsu.size(idx);
            int nt = -1;
            if (dsu.find(idx) != dsu.find(f[idx])) {
                nt = dsu.getType(f[idx]);
            }
            if (nt != -1) {
                ans[i][nt] += dsu.size(idx);
            }
            t[idx] = nt;
            dsu.setType(idx, -1);
            dsu.join(f[idx], idx);
        } else {
            ans[i][t[idx]] -= dsu.size(idx);
            int nxt = lower_bound(fa[idx].begin(), fa[idx].end(), i) - fa[idx].begin() - 1;
            t[idx] = upd[fa[idx][nxt]].second;
            dsu.setType(idx, t[idx]);
            ans[i][t[idx]] += dsu.size(idx);
        }
    }
    for (int i = 1; i <= m; i++) {
        for (int j = 0; j < 3; j++) {
            if (j) cout << ' ';
            cout << ans[i][j];
        }
        cout << '\n';
    }
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    solve();
    return 0;
}
