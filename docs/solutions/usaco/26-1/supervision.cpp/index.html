Analysis

the idea is that we keep a dp for all coaches. dp[j] = number of subsets that include the jth coach. for all dp[j], there is a set of campers that the jth coach can cover let the number of campers that coach j covers be cov(j) we get a basic recurrence dp[j] = 2^cov(j) * dp[i] for all i < j however, notice that this isn't fully accurate, because if coach j-1 covered a camper that coach j also covers, this camper is double counted to correct, we have dp[j] = 2^cov(j) * dp[i]/2^overlap(i, j) for all i < j overlap (i, j) is the number of campers that coaches i and j share coverage in so then the question is how we can update this DP quickly doing so naively is of course o(n^2), and can be done as follows for (int i = 1; i <= c.size(); i++) { for (int j = i - 1; j >= 0; j--) { int r = (j ? mp(c[j - 1] + d) : 0); int l = mp(c[i - 1]); int overlap = max(0, countcampers(l, r)); dp[i] += p2[cov(l) - overlap] * dp[j]; dp[i] %= MOD; } ans += dp[i]; } so how does o(n^2) become o(nlogn)? the trick is that if we order the cows and coaches left to right, processing by index, if the current index is a camper, any coach to the right will no longer ever be able to cover the current index. as we move left to right, overlap (i, x) is monotonically decreasing and it decreases specifically when the processed index moves past a covered cow therefore, maintain a segment tree of dp[i]/overlap(i, x), where overlap(i, x) is initialized to be cov(i) for the current coach if the processed cow is a coach, calculate dp[j] with the formula otherwise, it's a camper, and we update dp[i]/2^overlap(i, x) by multiplying the elements by 2. we use coordinate compression to ensure that all indexes stay in O(n) to save memory for our segment tree important edge case is if coach[0] is at position 0. That's why we push (-1) into the coordinate compression: to ensure no collision here

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#ifdef LOCAL
#include "algo/debug.h"
#else
#define dbg(...) 0
#define dbgn(...) 0
#endif

const ll MOD = 1e9 + 7;

struct SegTree {
    int n;
    vector<long long> tree, lazy;

    SegTree(int n) : n(n), tree(4 * n, 0), lazy(4 * n, 1) {}

    void apply(int v, long long x) {
        tree[v] = tree[v] * x % MOD;
        lazy[v] = lazy[v] * x % MOD;
    }

    void push(int v) {
        if (lazy[v] != 1) {
            apply(v * 2, lazy[v]);
            apply(v * 2 + 1, lazy[v]);
            lazy[v] = 1;
        }
    }

    void set(int v, int tl, int tr, int i, long long x) {
        if (tl == tr) {
            tree[v] = (x % MOD + MOD) % MOD;
            lazy[v] = 1;
            return;
        }
        push(v);
        int tm = (tl + tr) / 2;
        if (i <= tm)
            set(v * 2, tl, tm, i, x);
        else
            set(v * 2 + 1, tm + 1, tr, i, x);
        tree[v] = (tree[v * 2] + tree[v * 2 + 1]) % MOD;
    }

    void dbl(int v, int tl, int tr, int l, int r) {
        if (l <= tl && tr <= r) {
            apply(v, 2);
            return;
        }
        push(v);
        int tm = (tl + tr) / 2;
        if (l <= tm) dbl(v * 2, tl, tm, l, r);
        if (r > tm) dbl(v * 2 + 1, tm + 1, tr, l, r);
        tree[v] = (tree[v * 2] + tree[v * 2 + 1]) % MOD;
    }

    long long sum(int v, int tl, int tr, int l, int r) {
        if (l <= tl && tr <= r) return tree[v];
        push(v);
        int tm = (tl + tr) / 2;
        long long ans = 0;
        if (l <= tm) ans += sum(v * 2, tl, tm, l, r);
        if (r > tm) ans += sum(v * 2 + 1, tm + 1, tr, l, r);
        return ans % MOD;
    }

    void set(int i, long long x) { set(1, 0, n - 1, i, x); }
    void dbl(int l, int r) { dbl(1, 0, n - 1, l, r); }
    long long sum(int l, int r) { return sum(1, 0, n - 1, l, r); }
};

void solve() {
    int n, d;
    cin >> n >> d;
    vector<ll> cc;
    vector<pair<ll, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
        cc.push_back(v[i].first);
        cc.push_back(v[i].first + d);
    }
    cc.push_back(-1);
    sort(cc.begin(), cc.end());
    cc.erase(unique(cc.begin(), cc.end()), cc.end());
    vector<ll> p2(n + 1);
    p2[0] = 1;
    for (int i = 1; i <= n; i++) {
        p2[i] = p2[i - 1] * 2 % MOD;
    }
    vector<int> g;
    for (int i = 0; i < n; i++) {
        if (!v[i].second) g.push_back(v[i].first);
    }
    auto mp = [&](ll x) -> int { return lower_bound(cc.begin(), cc.end(), x) - cc.begin(); };
    vector<int> campers(cc.size());
    for (int i = 0; i < (int)g.size(); i++) {
        campers[mp(g[i])]++;
    }
    for (int i = 1; i < (int)cc.size(); i++) {
        campers[i] += campers[i - 1];
    }
    auto countcampers = [&](int l, int r) -> int {
        return max(0, campers[r] - (l ? campers[l - 1] : 0));
    };
    auto cov = [&](int x) -> int { return countcampers(x, mp(cc[x] + d)); };

    SegTree tree(cc.size());
    tree.set(0, 1);
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        int idx = mp(v[i].first);
        if (v[i].second) {
            ll curr = tree.sum(0, idx);
            tree.set(idx, curr);
            ans += curr * p2[cov(idx)] % MOD;
            ans %= MOD;
        } else {
            int pidx = mp(v[i].first - d);
            tree.dbl(pidx, idx);
            tree.set(0, 1);
        }
    }
    cout << ans << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    solve();
    return 0;
}
