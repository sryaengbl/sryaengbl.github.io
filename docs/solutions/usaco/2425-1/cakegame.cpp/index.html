Analysis

key observation is to realize that the cakes elsie eats must be a continuous prefix and a continuous suffix the number of cakes that elsie gets is (n-1)/2, with floor division if elsie desires a certain prefix + suffix, she just needs to eat the prefix first and the suffix second. if bessie merges cakes, the worst case scenario for elsie is that bessie doesn't help her. moreover, bessie can always make a "special" cake in the center with the cakes that elsie won't choose. bessie can fuse the middle cake with a cake on the left. afterward, if elsie chooses a cake on the left, bessie fuses the special cake with the cake on its right if elsie chooses from the left, bessie fuses with the right. thus, bessie always has a strategy to prevent her special cake from being consumed by elsie.

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#ifdef LOCAL
#include "algo/debug.h"
#else
#define dbg(...) 0
#define dbgn(...) 0
#endif

void solve() {
    int n;
    cin >> n;
    vector<ll> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        a[i] += a[i - 1];
    }
    ll ans = 0;
    int len = (n - 1) / 2;
    for (int i = 0; i <= len; i++) {
        ans = max(ans, a[i] + a[n] - a[n - len + i]);
    }
    cout << a[n] - ans << ' ' << ans << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    int tt;
    cin >> tt;
    while (tt--) {
        solve();
    }
    return 0;
}
