#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int N = 1e5 + 5;

int n;
int p[N];
int ans[N];
vector<int> a[N];

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    cin >> n;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        a[x].push_back(i);
    }
    auto check = [&](int idx, int jump) -> int {
        int ret = 0, at = 0;
        while (at < a[idx].size()) {
            ret++;
            at = upper_bound(a[idx].begin(), a[idx].end(), a[idx][at] + jump) - a[idx].begin();
        }
        return ret;
    };
    for (int i = 1; i <= n; i++) {
        int c = a[i].size(), mn = 1;
        while (c > 1) {
            int low = mn, high = a[i].back() - a[i][0];
            while (low < high) {
                int mid = low + (high - low) / 2;
                if (check(i, mid) < c)
                    high = mid;
                else
                    low = mid + 1;
            }
            int nw = check(i, low);
            p[low] -= c - nw;
            c = nw;
            mn = low;
        }
    }
    p[0] = n;
    for (int i = 1; i <= n; i++) {
        p[i] += p[i - 1];
        cout << p[i] << '\n';
    }
    return 0;
}
