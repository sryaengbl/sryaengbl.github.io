#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#ifdef LOCAL
#include "algo/debug.h"
#else
#define dbg(...) 0
#define dbgn(...) 0
#endif

const ll MOD = 1e9 + 7;

void solve() {
    int n;
    string str;
    cin >> n >> str;
    vector<ll> dp(n + 1, 0);
    dp[0] = 1;
    vector<int> nxtB(n + 2, n + 1);
    for (int i = n; i >= 1; i--) {
        nxtB[i] = (str[i - 1] == 'B' ? i : nxtB[i + 1]);
    }
    vector<vector<ll>> bit(2, vector<ll>(n + 2));
    auto add = [&](int p, int i, ll v) {
        v = (v % MOD + MOD) % MOD;
        for (++i; i <= n + 1; i += i & -i) {
            bit[p][i] = (bit[p][i] + v) % MOD;
        }
    };
    auto pref = [&](int p, int i) {
        ll res = 0;
        for (++i; i > 0; i -= i & -i) {
            res = (res + bit[p][i]) % MOD;
        }
        return res;
    };
    auto sum = [&](int p, int l, int r) {
        if (l > r) return 0LL;
        return (pref(p, r) - (l ? pref(p, l - 1) : 0) + MOD) % MOD;
    };
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    auto insert = [&](int j) {
        add(j % 2, j, dp[j]);

        int b = nxtB[j + 1];
        if (b <= n) {
            pq.push({2 * b - j, j});
        }
    };
    insert(0);
    int lastR = 0;
    for (int i = 1; i <= n; i++) {
        if (str[i - 1] == 'R') lastR = i;
        while (!pq.empty() && pq.top().first <= i) {
            int j = pq.top().second;
            pq.pop();
            add(j % 2, j, -dp[j]);
        }
        if (str[i - 1] == 'X') {
            dp[i] = dp[i - 1];
        }
        int l = max(0, 2 * lastR - i);
        int r = i - 2;
        dp[i] = (dp[i] + sum(i % 2, l, r)) % MOD;
        insert(i);
    }
    cout << dp[n] << '\n';
}

int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    solve();
    return 0;
}
